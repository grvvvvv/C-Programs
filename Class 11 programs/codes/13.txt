#include <iostream>
#include <string>
using namespace std;
int main()
{
    string inputString;
    bool isPalindrome = true;
    cout << "Enter a string: ";
    cin >> inputString;
    int start = 0;
    int end = inputString.length() - 1;
    while (start < end)
    {
        if (tolower(inputString[start]) != tolower(inputString[end]))
        {
            isPalindrome = false;
            break;
        }
        start++;
        end--;
    }
    if (isPalindrome)
    {
        cout << "The string is a palindrome." << endl;
    }
    else
    {
        cout << "The string is not a palindrome." << endl;
    }
    return 0;
}
