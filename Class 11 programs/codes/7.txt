#include <iostream>
using namespace std;
int main()
{
    const int numEmployees = 20;
    double monthlySalaries[numEmployees];
    int count = 0;
    cout << "Enter the monthly salaries of " << numEmployees << " employees (in ₹):" << endl;
    for (int i = 0; i < numEmployees; i++)
    {
        cout << "Employee " << i + 1 << ": ";
        cin >> monthlySalaries[i];
        while (monthlySalaries[i] < 0)
        {
            cout << "Monthly salary cannot be negative. Please re-enter: ";
            cin >> monthlySalaries[i];
        }
    }
    for (int i = 0; i < numEmployees; i++)
    {
        double annualSalary = monthlySalaries[i] * 12;
        if (annualSalary > 100000)
        {
            count++;
        }
    }
    cout << "\nThe number of employees earning more than ₹1 lakh per annumis: " << count << endl;
    return 0;
}
