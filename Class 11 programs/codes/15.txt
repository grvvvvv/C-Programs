#include <iostream>
#include <sstream>
#include <string>
using namespace std;
int main()
{
    string sentence, word, longestWord = "";
    int maxLength = 0;
    cout << "Enter a sentence: ";
    getline(cin, sentence);
    stringstream ss(sentence);
    while (ss >> word)
    {
        if (word.length() > maxLength)
        {
            longestWord = word;
            maxLength = word.length();
        }
    }
    cout << "The longest word is: " << longestWord << endl;
    return 0;
}
