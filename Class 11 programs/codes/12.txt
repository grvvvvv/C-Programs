#include <iostream>
#include <string>
using namespace std;
int main()
{
    string inputString;
    int vowelsCount = 0, consonantsCount = 0;
    cout << "Enter a string: ";
    cin >> inputString;
    for (int i = 0; i < inputString.length(); i++)
    {
        char ch = tolower(inputString[i]);
        if (ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u')
        {
            vowelsCount++;
        }
        else if ((ch >= 'a' && ch <= 'z'))
        {
            consonantsCount++;
        }
    }
    cout << "Vowels: " << vowelsCount << ", Consonants: " << consonantsCount << endl;
    return 0;
}
