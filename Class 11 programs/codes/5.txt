#include <iostream>
using namespace std;
int main()
{
    int n;
    char ch;
    do
    {
        cout << "Enter a positive integer to calculate its factorial: ";
        cin >> n;
        unsigned long long fact = 1;
        for (int i = 1; i <= n; i++)
        {
            fact *= i;
        }
        cout << "Factorial = " << fact << endl;
        cout << "Repeat? (y/n): ";
        cin >> ch;
    } while (ch == 'y' || ch == 'Y');
    return 0;
}
