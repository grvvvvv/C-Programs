#include <iostream>
using namespace std;
int main()
{
    const int salesmen = 5;
    const int months = 12;
    double sales[salesmen][months];
    double totalSales[salesmen] = {0};
    cout << "Enter the sales data for " << salesmen << " salesmen for " << months << " months:\n";
    for (int i = 0; i < salesmen; i++)
    {
        cout << "\nSalesman " << i + 1 << ":" << endl;
        for (int j = 0; j < months; j++)
        {
            cout << "Month " << j + 1 << ": rs";
            cin >> sales[i][j];
            while (sales[i][j] < 0)
            {
                cout << "Sales cannot be negative. Please re-enter for Month " << j + 1 << ": ₹";
                cin >> sales[i][j];
            }
            totalSales[i] += sales[i][j];
        }
    }
    cout << "\nTotal sales made by each salesman:\n";
    for (int i = 0; i < salesmen; i++)
    {
        cout << "Salesman " << i + 1 << ": rs" << totalSales[i] << endl;
    }
    return 0;
}
